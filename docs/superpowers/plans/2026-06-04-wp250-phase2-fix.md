```python
    def _tt_tap_create_button(self, ui: str, cfg: dict) -> bool:
        """WP#250: тап НАСТОЯЩЕЙ TikTok create-«+», исключая профильную
        «Создать историю» (substring-коллизия 'Создать'∈'Создать историю' уводит
        на SocialMediaPickerActivity / share-to-story).

        Алгоритм: среди clickable-нод, чей (text+desc) матчит любой create-паттерн
        (cfg['plus_button']['desc']) И НЕ содержит ни одного cfg['create_exclude'],
        выбираем самую нижнюю (max cy) — настоящий «+» в нижней навигации, а не
        story-CTA в шапке. Тапаем её центр. False — если подходящих нод нет.
        """
        import xml.etree.ElementTree as ET, re
        descs = [d.lower() for d in cfg['plus_button']['desc']]
        excl = [e.lower() for e in cfg.get('create_exclude', [])]
        best = None  # (cy, cx)
        try:
            for n in ET.fromstring(ui).iter('node'):
                if n.get('clickable') != 'true':
                    continue
                label = ((n.get('text') or '') + ' ' + (n.get('content-desc') or '')).strip().lower()
                if not label:
                    continue
                if not any(p in label for p in descs):
                    continue
                if any(e in label for e in excl):
                    continue
                m = re.search(r'\[(\d+),(\d+)\]\[(\d+),(\d+)\]', n.get('bounds', ''))
                if not m:
                    continue
                cx = (int(m.group(1)) + int(m.group(3))) // 2
                cy = (int(m.group(2)) + int(m.group(4))) // 2
                if best is None or cy > best[0]:
                    best = (cy, cx)
        except Exception:
            return False
        if best is None:
            return False
        self.p.adb_tap(best[1], best[0])
        return True
```
