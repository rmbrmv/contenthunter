```python
    def _handle_tt_amplify_modal(self, ui_xml: str, wait: int):
        """[WP #118 r2] dismiss TikTok Amplify модала с cap → inferred_success.

        Tri-state (как _handle_tt_promo_inbox_modal):
          - True              — handled (caller: sleep + continue)
          - False             — irrecoverable (caller: abort wait_upload)
          - 'inferred_success'— cap exceeded; модал = доказательство
                                post-publish state → caller upload_confirmed + break.
        """
        self._amplify_iter += 1
        n = self._amplify_iter
        if n == 1:
            self.log_event(
                'info', 'TikTok: Amplify промо-модал обнаружен',
                meta={'category': 'tt_amplify_modal_detected',
                      'platform': self.platform, 'step': 'wait_upload',
                      'wait_iter': wait}
            )
        if n > self.MAX_TT_AMPLIFY_ITERATIONS:
            self.log_event(
                'info',
                f'TikTok: Amplify модал loop > {self.MAX_TT_AMPLIFY_ITERATIONS} '
                f'iter — inferred post-publish success',
                meta={'category': 'tt_post_publish_inferred_from_amplify_loop',
                      'platform': self.platform, 'step': 'wait_upload',
                      'iterations': n, 'wait_iter': wait}
            )
            return 'inferred_success'
        tapped = self.tap_element(
            ui_xml or '', self._TT_AMPLIFY_CLOSE_LABELS,
            exact=True, clickable_only=True,
        )
        strategy = 'not_now_tap' if tapped else 'back_keycode'
        if not tapped:
            try:
                self.adb('input keyevent KEYCODE_BACK')
            except Exception:
                pass
        self.log_event(
            'info',
            f'TikTok: Amplify dismiss attempt {n} via {strategy}',
            meta={'category': 'tt_amplify_modal_dismissed',
                  'platform': self.platform, 'step': 'wait_upload',
                  'iteration': n, 'strategy': strategy, 'wait_iter': wait}
        )
        return True
```

```python
    def _yt_restart_upload_activity(self, remote_media_path: str) -> bool:
        """[WP #117 r2] Перезапустить Shell_UploadActivity тем же файлом, когда
        foreground ушёл из YouTube (drift в профиль/лаунчер). Извлечено из
        inline-логики stuck-counter'а для переиспользования desc-trap-гардом.

        Returns True если intent отправлен (получили content-uri), иначе False.
        """
        self.adb('input keyevent KEYCODE_HOME')
        time.sleep(1)
        yt_pkg = self.platform_cfg.get('package', 'com.google.android.youtube')
        content_uri = self._get_mediastore_content_uri(remote_media_path)
        if not content_uri:
            log.warning('  YT: restart_upload_activity — нет content-uri, skip')
            return False
        self.adb(
            f'am start -n {yt_pkg}/com.google.android.apps.youtube.app.application.Shell_UploadActivity '
            f'-a android.intent.action.SEND -t "video/mp4" '
            f'--eu android.intent.extra.STREAM "{content_uri}" '
            f'--grant-read-uri-permission'
        )
        time.sleep(5)
        log.info('  YT: перезапустили Shell_UploadActivity после drift')
        return True
```
